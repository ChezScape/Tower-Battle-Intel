import fs from "node:fs";
import assert from "node:assert/strict";

const css = fs.readFileSync("desktop.css", "utf8");
const mobile = fs.readFileSync("mobile.css", "utf8");
const config = fs.readFileSync("config/appConfig.js", "utf8");

assert.match(config, /version:\s*"v4\.11i"/);
assert.match(css, /v4\.11i Compact Desktop Top Polish/);
assert.match(css, /max-width:\s*1040px\) and \(min-width:\s*860px/);
assert.match(css, /\.tbi-run-strip[\s\S]*grid-template-columns:\s*1fr !important/);
assert.match(css, /\.tbi-run-card[\s\S]*min-height:\s*88px !important/);
assert.match(css, /\.tbi-vs-core[\s\S]*min-height:\s*42px !important/);
assert.match(css, /\.tbi-side-column[\s\S]*gap:\s*7px !important/);
assert.match(css, /\.concept5-actions button,[\s\S]*min-height:\s*42px !important/);
assert.equal(mobile.includes("v4.11i Compact Desktop Top Polish"), false);

console.log("current-v4.11i-compact-top-polish.test.mjs passed");
